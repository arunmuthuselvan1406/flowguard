-- RootCause: z-score of each sensor in the 60 min before a stoppage SELECT d.event_id, r.sensor, (AVG(r.value) - b.mu) / NULLIF(b.sd,0) AS z_score FROM core.downtime_events d JOIN core.sensor_readings r ON r.machine_id = d.machine_id AND r.event_ts BETWEEN ADD_MINUTES(d.start_ts,-60) AND d.start_ts JOIN core.baseline b ON b.machine_id = r.machine_id AND b.sensor = r.sensor GROUP BY d.event_id, r.sensor, b.mu, b.sd ORDER BY ABS(z_score) DESC;
